# ------------------------------------------------------------------------------
# 1. Device Identification & Runtime Environment
# ------------------------------------------------------------------------------
# active_dev_id defines the target hardware runtime profile.
# Allowed values:
#   - ANDROID_TERMUX_PRIMARY   : Primary ARM64 mobile environment (Termux + Android OS)
#   - LINUX_WORKSTATION        : Standard x86_64 / ARM64 desktop Linux environment
#   - WINDOWS_DEV_FALLBACK     : Windows runtime via WSL or native drive letters
active_dev_id="ANDROID_TERMUX_PRIMARY"

# env_loaded indicates the status of loaded system definitions.
# Expected values: "user_definitions_loaded", "complete_profile_loaded"
env_loaded="user_definitions_loaded"

# ------------------------------------------------------------------------------
# 2. Canonical Pointer Architecture (Zero-Secret Sources of Truth)
# ------------------------------------------------------------------------------
# Primary system definitions pointer (non-sensitive paths, tools, libraries)
DEFINITIONS_FILE="/storage/emulated/0/.env/user/definitions.json"

# Secondary credentials store pointer (API keys, authentication secrets)
# WARNING: Kept strictly local; never committed to public version control.
SECURE_ENV_FILE="/storage/emulated/0/.env/secure/secureenv_file.json"

# Legacy fallback environment pointer
ENV_FILE_PATH="/storage/emulated/0/env_file.json"

# ------------------------------------------------------------------------------
# 3. Storage Hierarchy & Mount Priority Rules
# ------------------------------------------------------------------------------
# Platform-aware path resolution priority:
# 1. Primary Android Shared Storage : /storage/emulated/0
# 2. Removable External SD-Card    : /storage/[SDCARD_UUID] (Highest priority for media archives)
# 3. Termux POSIX Sandbox Root     : /data/data/com.termux/files/home
# 4. Desktop Linux Home Root       : ~/.local/share or ~/Admin
# 5. Windows Drive Letters         : C:/Admin or D:/Admin

ADMIN_ROOT="/storage/emulated/0/Admin"
ADMIN_DEV="/storage/emulated/0/Admin/Dev"
ADMIN_REPOS="/storage/emulated/0/Admin/Dev/Repos"
ADMIN_REPORTS="/storage/emulated/0/Admin/Resources/Reports"
GLOBAL_LOGS="/storage/emulated/0/Admin/System/data/logs"

# ------------------------------------------------------------------------------
# 4. Protected Directories & Governance Policies
# ------------------------------------------------------------------------------
# "only idx" (Index-Only Mode):
# In file-governor and archive routines, "only idx" instructs the scanner to
# build or update .path.bejson catalog tracking manifests without relocating,
# renaming, or mutating directory entries.
PROTECTED_PATHS_MODE="only_idx"

# Protected root paths where recursive deletions, moves, or wipes are blocked:
PROTECTED_ROOT_PATHS="/storage/emulated/0/Admin,/storage/emulated/0/Download"
